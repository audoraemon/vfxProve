extends SceneTree
## Runs one half of tests/run_all.gd's SUITES (v0.11 M4: the full suite can pass the 10-minute foreground cap): --half=1 the first
## half, --half=2 the rest. The suites call the same check() and near() as run_all.gd's.

var failures := 0
var checks := 0


func check(cond: bool, msg: String) -> void:
	checks += 1
	if not cond:
		failures += 1
		printerr("FAIL: ", msg)


func near(a: float, b: float, eps: float, msg: String) -> void:
	check(absf(a - b) <= eps, "%s (got %f expected %f)" % [msg, a, b])


func _initialize() -> void:
	var all: Array = (load("res://tests/run_all.gd") as GDScript).get_script_constant_map()["SUITES"]
	var half := 1
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--half="):
			half = int(a.trim_prefix("--half="))
	var mid := all.size() / 2
	for path in (all.slice(0, mid) if half == 1 else all.slice(mid)):
		var suite: GDScript = load(path)
		if suite == null or not suite.can_instantiate():
			check(false, "suite failed to load: %s" % path)
			continue
		var before := checks
		suite.run(self)
		if checks == before:
			check(false, "suite ran no checks (runtime error?): %s" % path)
	print("checks=%d failures=%d half=%d" % [checks, failures, half])
	quit(1 if failures > 0 else 0)

## The night's setup (v0.11 M4): the plan; each step's party set down on its ring and kept there on duty; the head and his escorts
## indoors, his route; each step and the head's time on the strip.
func _begin() -> void:
	_plan()
	for st in steps:
		st.spot = _walkable(st.spot)
	for i in steps.size():
		var picked := _appoint_party(i)
		for k in picked.size():
			var g := Guard.new()
			g.post = _party_post(i, k)
			g.man = picked[k]
			_set_down(picked[k], g.post)
			picked[k].go_duty(g.post)
			steps[i].party.append(g)
	if has_head:
		head = _appoint_head()
		escorts = _appoint_escorts()
		for v: Variant in [head] + escorts:
			if _alive(v):
				(v as Person).pace *= head_pace
		head_route = _head_route()
		_stop_left = head_stop
	crowd._field.enemy_killed.connect(_on_killed)
	timeline = _new_timeline()
	timeline.fired.connect(func(_id: String, label: String) -> void: rules.banner.emit(label.to_upper()))
	for i in steps.size():
		timeline.add(steps[i].at, "step_%d" % i, _step_label(i), _raise.bind(i), func() -> bool: return not steps[i].raised)
	if has_head and head_at >= 0.0:
		timeline.add(head_at, "head", _head_label(), _call_head, func() -> bool: return not head_out and not head_down())
	rules.banner.emit(_opening_banner())

		return false

## Everyone at the hunt's work (v0.11 M4): every guard, the head and his escorts (untyped), for a new one to leave out.
func _in_use() -> Array:
	var out: Array = []
	for st in steps:
		for g: Guard in st.party:
			out.append(g.man)
	if head != null:
		out.append(head)
	out.append_array(escorts)
	return out


## One step (v0.11 M4): the timeline and the chain; every frame each unbroken step's party and the head's search; every TICK the
## escorts' places; then whether the hunt has found its quarry.
func step(delta: float) -> void:
	_clock += delta
	timeline.step(delta)
	if found:
		return
	_chain()
	_tick_in -= delta
	var looking := _tick_in <= 0.0
	if looking:
		_tick_in = TICK
	for i in steps.size():
		if not steps[i].broken:
			_keep_party(i, delta, looking)
	_step_head(delta)
	if looking:
		_keep_escorts()
	if not found and _found():
		found = true
		rules.banner.emit(_found_banner())


## The second of the night step `i`'s standard is raised at the latest (v0.11 M4): its own time, or `chain` after the step before is
## broken if that is sooner. The step before still standing never brings it forward.
func step_due(i: int) -> float:
	var due := steps[i].at * timeline.stretch_factor()
	if i > 0 and steps[i - 1].broken:
		due = minf(due, steps[i - 1].broken_at + chain)
	return due


## The second of the night the head comes out at the latest (v0.11 M4): his time, or `chain` after the last step is broken.
func head_due() -> float:
	var due := head_at * timeline.stretch_factor()
	if not steps.is_empty() and steps.back().broken:
		due = minf(due, (steps.back() as HuntStep).broken_at + chain)
	return due


## Seconds until step `i`'s standard is raised at the latest (v0.11 M4); 0 once it is.
func due_in(i: int) -> float:
	return 0.0 if steps[i].raised else maxf(step_due(i) - _clock, 0.0)


## Each step whose chained time has come is raised; the head comes out once called and every standard is raised (v0.11 M4).
func _chain() -> void:
	for i in range(1, steps.size()):
		if not steps[i].raised and steps[i - 1].broken and _clock >= step_due(i):
			rules.banner.emit(_step_label(i).to_upper())
			_raise(i)
	if has_head and not head_out and not head_down() and _all_raised():
		var last: HuntStep = steps.back() if not steps.is_empty() else null
		if _head_called or (last != null and last.broken and _clock >= head_due()):
			_head_go(true)


## Every step's standard is raised (v0.11 M4).
func _all_raised() -> bool:
	for st in steps:
		if not st.raised:
			return false
	return true


## Step `i`'s standard is raised (v0.11 M4): shielded while guarded (_shield()) and watched for its fall. One that cannot be made
## counts the step broken, so the night never locks.
func _raise(i: int) -> void:
	var st := steps[i]
	if st.raised:
		return
	st.raised = true
	st.raised_at = _clock
	var s := _raise_standard(i)
	if s == null:
		push_warning("HuntDirector: no standard for step %d; counted broken" % i)
		_break(i)
		return
	st.standard = s
	s.damage_filter = _shield
	if not s.broken.is_connected(_on_broken):
		s.broken.connect(_on_broken)
	standards.append(s)


## The step whose standard `s` is, or -1 (v0.11 M4).
func _step_of(s: Structure) -> int:
	for i in steps.size():
		if is_instance_valid(steps[i].standard) and steps[i].standard == s:
			return i
	return -1


## Structure.damage_filter for a standard (v0.11 M4, Broken Lanterns' guarded shrine): guarded, the blow only shakes it; else it
## takes the blow as any building does.
func _shield(s: Structure, amount: float, source: Vector2, kind: StringName) -> void:
	var i := _step_of(s)
	if i >= 0 and _guarded(i):
		s.shake(SHAKE)
		return
	s.hp -= amount
	s.mark_hit(amount / s.max_hp, kind)
	if s.hp <= 0.0:
		s.destroy(source, kind)
	elif s.hp < s.max_hp * Structure.CRACK_AT:
		s.crack()


## A standard fell (v0.11 M4).
func _on_broken(s: Structure) -> void:
	var i := _step_of(s)
	if i >= 0 and not steps[i].broken:
		_break(i)


## Step `i` is broken for good (v0.11 M4): its party goes home, and the strip shows the next item's chained time.
func _break(i: int) -> void:
	var st := steps[i]
	st.broken = true
	st.broken_at = _clock
	rules.banner.emit(_broken_banner(i))
	for g: Guard in st.party:
		_release(g)
	if i + 1 < steps.size():
		timeline.expect("step_%d" % (i + 1), step_due(i + 1))
	elif has_head:
		timeline.expect("head", head_due())


## A guard of a broken step goes home (v0.11 M4, Spoiled Harvest's _release()): one on duty stood down, one calm sent home.
func _release(g: Guard) -> void:
	if not _alive(g.man):
		return
	var p := g.man as Person
	if p.mind == Person.Mind.DUTY:
		crowd.off_duty(p)
	elif p.mind in RECALL_MINDS and p.profile != null:
		p.regroup(p.profile.home)


## Guard `g` of step `i` stands guard now (v0.11 M4): alive, out of doors, not held by the god (BLIND), and within `guard_reach` of
## the standard -- of the spot before it is raised, or once it is down.
func _guarding(i: int, g: Guard) -> bool:
	if not _alive(g.man):
		return false
	var p := g.man as Person
	if p.inside or p.mind in BLIND:
		return false
	var st := steps[i]
	var s: Variant = st.standard
	var gap := (s as Structure).distance_to(p.ground_pos) if is_instance_valid(s) and not (s as Structure).destroyed \
		else st.spot.distance_to(p.ground_pos)
	return gap <= guard_reach


## Step `i`'s party at their work (v0.11 M4): a guard standing guard is kept on duty at his post (back on his feet within reach, he is
## put on duty again, so the town's regroup and evacuation pass him by); one off his post but alive is sent back `return_after` after
## he left (once back on his feet); a fallen one is replaced `relief_after` after he fell.
func _keep_party(i: int, delta: float, looking: bool) -> void:
	for g: Guard in steps[i].party:
		if _guarding(i, g):
			g.away = 0.0
			g.fallen = 0.0
			g.relief = false
			if looking and (g.man as Person).mind != Person.Mind.DUTY:
				_recall_to(g.man, g.post)
		elif _alive(g.man):
			g.fallen = 0.0
			if (g.man as Person).inside:
				continue
			g.away += delta
			if looking and g.away >= return_after:
				_recall_to(g.man, g.post)
		else:
			g.fallen += delta
			if looking and g.fallen >= relief_after:
				_relieve(i, g)


## A relief for a fallen guard of a step (v0.11 M4): the nearest lay citizen with a way to the post, dressed as a guard. The step's
## index is passed for a subclass that relieves by step.
func _relieve(_i: int, g: Guard) -> void:
	var p := _lay_reaching(g.post, _in_use())
	if p == null:
		return
	_dress_guard(p)
	g.man = p
	g.fallen = 0.0
	g.away = 0.0
	g.relief = true
	p.go_duty(g.post)
	rules.banner.emit("A NEW GUARD TAKES THE POST")


## The head's time has come (v0.11 M4, the timeline): out now if every standard is raised, else as soon as the last is.
func _call_head() -> void:
	_head_called = true
	if _all_raised():
		_head_go(false)


## The head comes out at `head_door` with his escorts and walks his route (v0.11 M4); `announce` gives the banner (the timeline
## gives its own).
func _head_go(announce: bool) -> void:
	if head_out:
		return
	head_out = true
	head_out_at = _clock
	if announce:
		rules.banner.emit(_head_label().to_upper())
	if not _alive(head):
		return
	var p := head as Person
	var door := _walkable(head_door) if head_door != Vector2.INF else p.ground_pos
	if p.inside:
		_bring_out(p, door)
	if not head_route.is_empty():
		p.go_duty(head_route[0])
	for k in escorts.size():
		var v: Variant = escorts[k]
		if _alive(v) and (v as Person).inside:
			_bring_out(v as Person, _walkable(door + Vector2(ESCORT_SIDE * (-1.0 if k % 2 == 0 else 1.0), -escort_gap)))


## The head's search, every frame (v0.11 M4): on duty, at his point he stops `head_stop` (searching from the first door), then walks
## on to the next; knocked off his errand, he takes it up again back on his feet, at the door he had reached, his stop begun anew.
func _step_head(delta: float) -> void:
	if not head_out or found or head_down() or head_route.is_empty():
		return
	var p := head as Person
	if p.inside:
		return
	var goal := head_route[_route_at]
	if p.mind == Person.Mind.DUTY:
		if p.ground_pos.distance_to(goal) <= ARRIVE:
			searching = true
			if _route_at >= head_route.size() - 1:
				return
			_stop_left -= delta
			if _stop_left <= 0.0:
				_route_at += 1
				_stop_left = head_stop
				p.go_duty(head_route[_route_at])
		elif p.anchor.distance_to(goal) > MOVE:
			p.go_duty(goal)
	elif p.mind in RECALL_MINDS:
		_stop_left = head_stop
		p.go_duty(goal)


## The escorts keep `escort_gap` behind the head, ESCORT_SIDE to either side of his track (v0.11 M4): beyond one Silent Doom of him.
## One frightened, held or fighting is left be, and walks back once on his feet.
func _keep_escorts() -> void:
	if not head_out or not _alive(head) or (head as Person).inside:
		return
	var p := head as Person
	var goal := head_route[mini(_route_at, head_route.size() - 1)] if not head_route.is_empty() else p.ground_pos
	var dir := goal - p.ground_pos
	dir = dir.normalized() if dir.length() > 0.01 else Vector2.DOWN
	var side := Vector2(-dir.y, dir.x)
	for k in escorts.size():
		var v: Variant = escorts[k]
		if not _alive(v) or (v as Person).inside:
			continue
		var e := v as Person
		var spot := _walkable(p.ground_pos - dir * escort_gap + side * ESCORT_SIDE * (-1.0 if k % 2 == 0 else 1.0))
		if e.mind in RECALL_MINDS or (e.mind == Person.Mind.DUTY and e.anchor.distance_to(spot) > 0.5):
			e.go_duty(spot)


## A death (v0.11 M4): the head's, any death, ends him for the objective; his escorts go home. Nobody takes up the search.
func _on_killed(e: DummyEnemy, _kind: StringName) -> void:
	if head == null or not is_instance_valid(head) or e != head or head_fell:
		return
	head_fell = true
	for v: Variant in escorts:
		if _alive(v):
			crowd.off_duty(v as Person)


## The head is down (v0.11 M4, decision 113): there is none, he was killed, or his body is gone.
func head_down() -> bool:
	return not has_head or head_fell or not _alive(head)


## Step `i`'s standard, standing or fallen, or null (v0.11 M4).
func standard(i: int) -> Structure:
	var s: Variant = steps[i].standard
	return s as Structure if is_instance_valid(s) else null


## Step `i`'s standard stands raised and guarded now (v0.11 M4). Only reads.
func guarded(i: int) -> bool:
	return steps[i].raised and not steps[i].broken and standard(i) != null and _guarded(i)


## Step `i`'s guards standing guard now (v0.11 M4). Only reads.
func party_on(i: int) -> Array[Person]:
	var out: Array[Person] = []
	for g: Guard in steps[i].party:
		if _guarding(i, g):
			out.append(g.man as Person)
	return out


## A relief of step `i` walking in, or null (v0.11 M4, for the scripted player). Only reads.
func relief_walking(i: int) -> Person:
	for g: Guard in steps[i].party:
		if g.relief and _alive(g.man) and not _guarding(i, g) and not (g.man as Person).inside:
			return g.man as Person
	return null


## Whether step `i` is broken (v0.11 M4).
func broken(i: int) -> bool:
	return steps[i].broken


## How many steps are broken (v0.11 M4).
func broken_count() -> int:
	var n := 0
	for st in steps:
		n += 1 if st.broken else 0
	return n


## Every step is broken (v0.11 M4); false with none.
func broken_all() -> bool:
	return not steps.is_empty() and broken_count() == steps.size()


## The current step (v0.11 M4): the first raised and unbroken, or -1.
func current() -> int:
	for i in steps.size():
		if steps[i].raised and not steps[i].broken:
			return i
	return -1


## The next step still to be raised, or -1 (v0.11 M4).
func next_step() -> int:
	for i in steps.size():
		if not steps[i].raised:
			return i
	return -1


## Why the night is lost now, "" while it is not (v0.11 M4): the hunt found its quarry.
func lost_reason() -> String:
	return lost_word if found else ""


## The map tags, most important first (v0.11 M4, mission spec §3):
## - each standing standard, gold; the current one pointed at, "- GUARDED" while guarded;
## - the next step's spot with its chained time, grey, pointed at;
## - the head waiting (grey, at his door), or out (gold, pointed at, "- SEARCHING" once he searches), and his escorts, red;
## - the quarry, steel blue, pointed at;
## - each unbroken step's guards: labelled on guard, a red diamond on a relief walking in, a small red mark on one away;
## - each standing standard's district outlined in red, on a red pip at its foot (decision 114).
func tags() -> Array[MapTag]:
	var out: Array[MapTag] = []
	var cur := current()
	for i in steps.size():
		var s := standard(i)
		if s == null or steps[i].broken or s.destroyed:
			continue
		var label := standard_label + (" - GUARDED" if i == cur and guarded(i) else "")
		out.append(MapTag.place(s.center(), MARK_TARGET, label, s.height, i == cur))
	var n := next_step()
	if n >= 0:
		out.append(MapTag.place(steps[n].spot, MARK_NOT_YET, "%s %s" % [next_label, UiTheme.clock(due_in(n))], 0.0, true))
	if has_head and _alive(head):
		var h := head as Person
		if not head_out and head_door != Vector2.INF:
			out.append(MapTag.place(_walkable(head_door), MARK_NOT_YET, head_waiting_label, 0.0, false))
		elif head_out and not h.inside:
			out.append(MapTag.person(h.ground_pos, MARK_TARGET, head_label + (" - SEARCHING" if searching else ""), true))
			for v: Variant in escorts:
				if _alive(v) and not (v as Person).inside:
					out.append(MapTag.person((v as Person).ground_pos, MARK_WATCHED, escort_label))
	var q := _quarry_point()
	if q != Vector2.INF:
		out.append(MapTag.place(q, MARK_QUARRY, quarry_label, 0.0, true))
	for i in steps.size():
		if steps[i].broken:
			continue
		for g: Guard in steps[i].party:
			if not _alive(g.man) or (g.man as Person).inside:
				continue
			var p := g.man as Person
			if _guarding(i, g):
				out.append(MapTag.person(p.ground_pos, MARK_WATCHED, guard_label))
			elif g.relief:
				out.append(MapTag.person(p.ground_pos, MARK_WATCHED))
			else:
				out.append(MapTag.pip(p.ground_pos, MARK_WATCHED))
	for i in steps.size():
		var s := standard(i)
		if s == null or steps[i].broken or s.destroyed or not steps[i].district.has_area():
			continue
		var o := MapTag.pip(s.center(), MARK_WATCHED)
		o.outline = steps[i].district
		out.append(o)
	return out


## The hint's phase (v0.11 M4, decision 110): "searching" or "captain" while the head is out; "guarded" or "open" for the current
## standard; "waiting" while none stands and a step is still to come; "" before the first is raised.
func hint_phase() -> String:
	if found:
		return ""
	if has_head and head_out and not head_down():
		return "searching" if searching else "captain"
	var cur := current()
	if cur >= 0:
		return "guarded" if guarded(cur) else "open"
	if broken_count() > 0 and next_step() >= 0:
		return "waiting"
	return ""


## What the results show (v0.11 M4): the steps broken, the head down, and whether the hunt found its quarry.
func report() -> Dictionary:
	return {"broken": broken_count(), "head_down": head_down(), "found": found}


## Lets go of every standard's filter and signal (a standard may be a town tower that outlives the night, decision 111), the kills,
## and the timeline (v0.11 M4). Safe to call twice.
func teardown() -> void:
	for v: Variant in standards:
		if is_instance_valid(v):
			var s := v as Structure
			s.damage_filter = Callable()

## Target `i`'s crew at its work (v0.11 M4). Each on guard is held there; one off his post but alive is sent back RETURN_AFTER after he
## left (a shift man waiting indoors is not away); a fallen one is replaced RELIEF_AFTER after he fell. With one or more on guard: a
## fallen target not in use is rebuilt (decision 101); a standing one is mended; one in use and alight is held to and beaten out after
## SMOTHER_AFTER, its burning forgotten.
func _work(i: int, delta: float, looking: bool) -> void:
	var s := targets[i]
	var on := 0
	for c: Crew in crews[i]:
		if _on_guard(i, c):
			on += 1
			c.away = 0.0
			c.fallen = 0.0
			c.relief = false
			if looking and (c.man as Person).mind != Person.Mind.DUTY:
				_recall_to(c.man, c.post)  # back on his feet within reach: on duty again, so the town's flight passes him by
		elif _alive(c.man):
			c.fallen = 0.0
			if (c.man as Person).inside:
				continue
			c.away += delta
			if looking and c.away >= RETURN_AFTER:
				_recall_to(c.man, c.post)
		else:
			c.fallen += delta
			if looking and c.fallen >= RELIEF_AFTER:
				_relieve(i, c)
	if not is_instance_valid(s):
		return
	if s.destroyed:
		_alight[i] = 0.0
		if in_use[i] or on == 0:
			return

			rebuild_progress[i] = 0.0

	if on == 0:
		_alight[i] = 0.0
		return

		return

		forget_burn(s)

## that kept him at his post while it burned ends with it.
func _release(c: Crew) -> void:

		return

		p.fearless_left = 0.0

## The map tags, most important first (v0.11 M4, mission spec §1): each target not razed, pointed at -- not in use, grey, "IN USE BY"
## its chained time, or "DOWN" (with its rebuild's seconds while one works on it); in use, gold, or orange while burning -- then each
## crewman: ENGINEER on guard, a red diamond on a relief walking in, a small red mark on one away.

			out.append(MapTag.place(s.center(), MARK_SEALED, label, s.height, true))

			out.append(MapTag.place(s.center(), MARK_BURNING, "%s - BURNING" % name, s.height, true))

			continue

func _bread_act(rules: Rules, slots: Dictionary, d: CrownsBreadDirector, crowd: Crowd) -> String:

		if d.in_use[i] and not d.razed(d.targets[i]):

			break

				return did

					rules.cast(slots.smite, s.center())
					return "smite the %s" % CrownsBreadDirector.WORDS[cur]
			elif cur == 2 and _slot_ready(rules, slots, "smite"):
				rules.cast(slots.smite, s.center())
				return "smite the bridge"
	var next := d.upcoming()

			if did != "":

func _bread_clear(rules: Rules, slots: Dictionary, d: CrownsBreadDirector, crowd: Crowd, i: int, on: Array[Person], smite_ok: bool) -> String:
	var s := d.targets[i]
	var word := String(CrownsBreadDirector.WORDS[i])
	if smite_ok and _slot_ready(rules, slots, "smite"):
		for p in on:
			if s.distance_to(p.ground_pos) <= SmiteFx.HIT_R:
				rules.cast(slots.smite, p.ground_pos)
				return "smite an engineer at the %s" % word

		var pick := on[0]
		for p in on:
			if _bread_alone(crowd, p):
				pick = p
				break
		rules.cast(slots.doom, pick.ground_pos)
		return "doom an engineer at the %s" % word

		for p in on:

				continue
			var off := p.ground_pos - s.center()
			var away := p.ground_pos + (off.normalized() if off.length() > 0.01 else Vector2.RIGHT) * BREAD_AWAY
			rules.cast(slots.whisper, p.ground_pos, {"target": p, "to": MindWhisperFx.clamp_to(crowd._grid, p.ground_pos, away)})
			return "whisper an engineer off the %s" % word
	return ""

## One look of The Prisoner's policy (v0.11 M4): what was cast, or "". The first that applies: bound, a gaoler within sight of her --
## Silent Doom on him (beyond its reach of her), else Mind Whisper sends him off; bound, no gaoler near -- Ember the stocks; held --
## Discord on her seizer, or Doom him once she stands clear; free -- a watcher (the postern's watch aside) near her or her next
## stretch, whisper him off, else Doom, else Discord; Mind Whisper sends her along her road, stopping short of a watched postern; at
## the postern, watched, its change more than PRISONER_SOON off -- whisper one of its watch off. Never a loud power.
func _prisoner_act(rules: Rules, slots: Dictionary, d: PrisonerDirector, crowd: Crowd) -> String:
	var her := d.charge
	if not WarningDirector._alive(her) or her.inside:
		return ""
	if d.bound:
		for v: Variant in d.gaolers:
			if not d._watching(v) or (v as Person).ground_pos.distance_to(her.ground_pos) > EscortDirector.SIGHT + PRISONER_WARN:

				rules.cast(slots.doom, g.ground_pos)
				return "doom a gaoler"
			if _slot_ready(rules, slots, "whisper") and not g.shaken():
				rules.cast(slots.whisper, g.ground_pos, {"target": g, "to": _prisoner_away(crowd, g, her.ground_pos)})
				return "whisper a gaoler off"
			return ""

				and _slot_ready(rules, slots, "ember"):
			rules.cast(slots.ember, d.stocks.center())
			return "ember the stocks"
		return ""

		var s := d.seizer as Person
		if _slot_ready(rules, slots, "discord"):
			rules.cast(slots.discord, s.ground_pos)

			rules.cast(slots.doom, s.ground_pos)
			return "doom her seizer"
		return ""

	var ahead := _road_ahead(road, her.ground_pos, PRISONER_STRETCH)
	for v: Variant in d._watchers():

			continue

		if not going and her.ground_pos.distance_to(to) > 1.0:

## Where Mind Whisper sends `p` (v0.11 M4): PRISONER_AWAY straight away from `from`, on ground the whisper can reach.
func _prisoner_away(crowd: Crowd, p: Person, from: Vector2) -> Vector2:
	var off := p.ground_pos - from
	return MindWhisperFx.clamp_to(crowd._grid, p.ground_pos, p.ground_pos + (off.normalized() if off.length() > 0.01 else Vector2.RIGHT) * PRISONER_AWAY)


## The point `dist` along polyline `road` beyond where `at` lies nearest it (v0.11 M4); its end if nearer.
func _road_ahead(road: Array, at: Vector2, dist: float) -> Vector2:
	var along := _road_progress(road, at) + dist
	var run := 0.0
	for k in range(road.size() - 1):
		var a: Vector2 = road[k]
		var b: Vector2 = road[k + 1]
		var seg := a.distance_to(b)
		if run + seg >= along:
			return a.lerp(b, (along - run) / maxf(seg, 0.001))
		run += seg
	return road.back()


## How far along polyline `road` the point nearest `at` lies (v0.11 M4).
func _road_progress(road: Array, at: Vector2) -> float:
	var best := INF
	var best_run := 0.0
	var run := 0.0
	for k in range(road.size() - 1):
		var a: Vector2 = road[k]
		var b: Vector2 = road[k + 1]
		var q := Geometry2D.get_closest_point_to_segment(at, a, b)
		if q.distance_to(at) < best:
			best = q.distance_to(at)
			best_run = run + a.distance_to(q)
		run += a.distance_to(b)
	return best_run


## How far `p` stands from the segment `a`-`b` (v0.11 M4).
func _seg_dist(p: Vector2, a: Vector2, b: Vector2) -> float:
	return p.distance_to(Geometry2D.get_closest_point_to_segment(p, a, b))

## Doom on him; a raised standard unguarded, Smite on its far side from its guards; each raised standard guarded, then the next sweep
## within HUNT_AHEAD -- Discord where it holds two or more of a raised standard's guards, else Silent Doom on a guard alone (else any),
## else Mind Whisper sends one off; a relief walking in, Silent Doom on him.
func _hunt_act(rules: Rules, slots: Dictionary, d: HereticHuntDirector, crowd: Crowd) -> String:
	if d.head_out and WarningDirector._alive(d.head) and not (d.head as Person).inside and _slot_ready(rules, slots, "doom"):
		rules.cast(slots.doom, (d.head as Person).ground_pos)

		var s := d.standard(i)
		if s == null or d.steps[i].broken or s.destroyed or d.guarded(i):
			continue
		if _slot_ready(rules, slots, "smite"):
			rules.cast(slots.smite, _hunt_far_side(d, i))
			return "smite the standard of %s" % d.steps[i].words
	var work: Array[int] = []
	for i in d.steps.size():
		if d.steps[i].raised and not d.steps[i].broken and d.guarded(i):
			work.append(i)
	var n := d.next_step()
	if n >= 0 and d.due_in(n) <= HUNT_AHEAD:
		work.append(n)
	for i in work:
		var on := d.party_on(i)
		if on.is_empty():
			continue

		if on.size() >= 2 and d.steps[i].raised and _slot_ready(rules, slots, "discord"):
			var at := _hunt_pair(on)
			if at != Vector2.INF:
				rules.cast(slots.discord, at)
				return "discord the guards of %s" % words
		if _slot_ready(rules, slots, "doom"):
			var pick := on[0]
			for p in on:
				if _bread_alone(crowd, p):
					pick = p
					break
			rules.cast(slots.doom, pick.ground_pos)
			return "doom a guard of %s" % words
		if _slot_ready(rules, slots, "whisper"):
			for p in on:
				if p.shaken():
					continue
				var off := p.ground_pos - d.steps[i].spot
				var away := p.ground_pos + (off.normalized() if off.length() > 0.01 else Vector2.RIGHT) * HUNT_AWAY
				rules.cast(slots.whisper, p.ground_pos, {"target": p, "to": MindWhisperFx.clamp_to(crowd._grid, p.ground_pos, away)})
				return "whisper a guard of %s off" % words
	for i in d.steps.size():
		if d.steps[i].broken:
			continue
		var r := d.relief_walking(i)
		if r != null and _slot_ready(rules, slots, "doom"):
			rules.cast(slots.doom, r.ground_pos)
			return "doom a relief"
	return ""


## Where one Discord holds the most of `on`, two at least (v0.11 M4): on one of them or midway between two; Vector2.INF for nowhere.
func _hunt_pair(on: Array[Person]) -> Vector2:
	var best := Vector2.INF
	var most := 1
	var spots: Array[Vector2] = []
	for a in on.size():
		spots.append(on[a].ground_pos)
		for b in range(a + 1, on.size()):
			spots.append((on[a].ground_pos + on[b].ground_pos) * 0.5)
	for at in spots:
		var held := 0
		for p in on:
			held += 1 if p.ground_pos.distance_to(at) <= DiscordFx.DISCORD_R - 0.1 else 0
		if held > most:
			most = held
			best = at
	return best


## A Smite's aim on standard `i` (v0.11 M4): HUNT_FAR from its centre, away from the nearest of its guards still standing.
func _hunt_far_side(d: HereticHuntDirector, i: int) -> Vector2:
	var s := d.standard(i)
	var near := Vector2.INF
	for g: HuntDirector.Guard in d.steps[i].party:
		if WarningDirector._alive(g.man) and (near == Vector2.INF
				or (g.man as Person).ground_pos.distance_to(s.center()) < near.distance_to(s.center())):
			near = (g.man as Person).ground_pos
	if near == Vector2.INF or near.distance_to(s.center()) < 0.01:
		return s.center()
	return s.center() + (s.center() - near).normalized() * HUNT_FAR

## The people and the buildings `d` has claimed (v0.11 M4).
static func _actors(d: MissionDirector) -> Dictionary:
	var people: Array = []
	var places: Array = []
	if d is CrownsBreadDirector:
		var cb := d as CrownsBreadDirector
		for list: Array in cb.crews:

		places.append(hh.cellar)

static func _reserved(t) -> void:
	for id: String in NEW:
		var a := Kit.world(TierBook.board(id))
		var first := _actors(a.d)
		var people_ix := []
		for p: Variant in first.people:
			people_ix.append((a.crowd as Crowd).citizens.find(p))
		var place_ix := []
		for s: Variant in first.places:
			place_ix.append((a.town as Town)._built.find(s))
		Kit.done(a)
		var hold := func(d: MissionDirector, rules: Rules) -> void:
			for i: int in people_ix:
				if i >= 0:
					d.reserved.append(rules.crowd().citizens[i])
			for i: int in place_ix:
				if i >= 0:
					d.reserved_places.append(rules.town()._built[i])
		var b := Kit.world(TierBook.board(id), null, hold)
		var second := _actors(b.d)
		var clash := []
		for p: Variant in second.people:
			if people_ix.has((b.crowd as Crowd).citizens.find(p)):
				clash.append(p)
		for s: Variant in second.places:
			if s != null and place_ix.has((b.town as Town)._built.find(s)):
				clash.append(s)
		t.check(clash.is_empty() and not second.people.is_empty(),
			"%s: the people and buildings a first night claimed, reserved, are never claimed again (%d clashes)" % [id, clash.size()])
		Kit.done(b)
		var heard := 0
		var hit := 0
		for night in 4:
			var descent := Descent.new().setup(TierBook.board(id), Descent.seed_for(night, id))
			var n := Kit.world(TierBook.board(id), null, func(d: MissionDirector, rules: Rules) -> void: descent.reserve(rules, d))
			var mine := _actors(n.d)
			for w in descent.wishes:
				heard += 1
				for p in w.people():
					hit += 1 if (mine.people as Array).has(p) else 0
				for st in w.places():
					hit += 1 if (mine.places as Array).has(st) else 0
			descent.release()
			Kit.done(n)
		t.check(heard > 0 and hit == 0, "%s: four seeded board nights' wishes, reserved first, never become its people or buildings" % id)
